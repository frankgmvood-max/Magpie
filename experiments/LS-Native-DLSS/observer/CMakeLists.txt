cmake_minimum_required(VERSION 3.20)
project(LSNativeObserver LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The observer requires a Windows x64 target")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LS_ENABLE_NGX "Build the experimental native NVIDIA backend" OFF)
set(NGX_ROOT "" CACHE PATH "Pinned NVIDIA DLSS SDK checkout")
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(MINHOOK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/Magpie.Core/third_party/minhook")
add_library(native_observer_minhook STATIC
  ${MINHOOK_ROOT}/src/buffer.c ${MINHOOK_ROOT}/src/hook.c
  ${MINHOOK_ROOT}/src/trampoline.c ${MINHOOK_ROOT}/src/hde/hde64.c)
target_include_directories(native_observer_minhook PUBLIC ${MINHOOK_ROOT}/include)
add_library(Lossless SHARED proxy.cpp observation_win.cpp)
if(NOT MSVC)
  target_sources(Lossless PRIVATE Lossless.def)
endif()
target_compile_definitions(Lossless PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Lossless PRIVATE native_observer_minhook d3d11 dxgi bcrypt)
if(LS_ENABLE_NGX)
  if(NOT EXISTS "${NGX_ROOT}/include/nvsdk_ngx_helpers_dlssg_d3d.h")
    message(FATAL_ERROR "NGX_ROOT must contain the pinned NVIDIA DLSS SDK")
  endif()
  set(NGX_LIB "${NGX_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
  if(NOT EXISTS "${NGX_LIB}")
    message(FATAL_ERROR "Missing static CRT NGX import library")
  endif()
  add_library(native-nvidia-backend STATIC ../backend/native_backend.cpp)
  target_include_directories(native-nvidia-backend PRIVATE "${NGX_ROOT}/include" ../third_party/nvof)
  target_compile_definitions(native-nvidia-backend PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(native-nvidia-backend PUBLIC "${NGX_LIB}" d3d11 d3d12 d3dcompiler dxgi
    shlwapi wintrust advapi32 ole32 version bcrypt shell32 user32)
  target_compile_definitions(Lossless PRIVATE LS_WITH_NGX)
  target_link_libraries(Lossless PRIVATE native-nvidia-backend)
  # Separate translation unit: the synthetic provider is absent from Lossless.dll.
  add_executable(native-backend-tests ../tests/windows/test_native_backend.cpp ../backend/native_backend.cpp)
  target_include_directories(native-backend-tests PRIVATE "${NGX_ROOT}/include" ../third_party/nvof)
  target_compile_definitions(native-backend-tests PRIVATE LS_NATIVE_TEST UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(native-backend-tests PRIVATE "${NGX_LIB}" d3d11 d3d12 d3dcompiler dxgi
    shlwapi wintrust advapi32 ole32 version bcrypt shell32 user32)
  add_executable(native-hook-tests ../tests/windows/test_observer_warp.cpp observation_win.cpp ../backend/native_backend.cpp)
  target_include_directories(native-hook-tests PRIVATE "${NGX_ROOT}/include" ../third_party/nvof)
  target_compile_definitions(native-hook-tests PRIVATE LS_OBSERVER_TEST LS_NATIVE_TEST LS_WITH_NGX UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(native-hook-tests PRIVATE native_observer_minhook "${NGX_LIB}" d3d11 d3d12 d3dcompiler dxgi
    shlwapi wintrust advapi32 ole32 version bcrypt shell32 user32)
  if(MSVC)
    target_compile_options(native-nvidia-backend PRIVATE /W4 /WX /EHsc)
    target_compile_options(native-backend-tests PRIVATE /W4 /WX /EHsc)
    target_compile_options(native-hook-tests PRIVATE /W4 /WX /EHsc)
  endif()
endif()
set_target_properties(Lossless PROPERTIES PREFIX "" OUTPUT_NAME "Lossless")
if(MSVC)
  target_compile_options(Lossless PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(Lossless PRIVATE -Wall -Wextra -Werror)
  target_link_options(Lossless PRIVATE -static-libgcc -static-libstdc++)
endif()
add_executable(observation-tests ../tests/test_observation.cpp)
add_executable(slot-policy-tests ../tests/test_slot_policy.cpp)
add_executable(source-pair-tests ../tests/test_source_pair.cpp)
add_library(fake-native SHARED ../tests/windows/fake_native.cpp)
set_target_properties(fake-native PROPERTIES PREFIX "" OUTPUT_NAME "Lossless_original"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/abi-fixture/$<CONFIG>")
add_dependencies(fake-native Lossless)
add_custom_command(TARGET fake-native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "$<TARGET_FILE:Lossless>" "$<TARGET_FILE_DIR:fake-native>/Lossless.dll")
add_executable(proxy-abi-tests ../tests/windows/test_proxy_abi.cpp)
add_executable(observer-warp-tests ../tests/windows/test_observer_warp.cpp observation_win.cpp)
target_compile_definitions(observer-warp-tests PRIVATE LS_OBSERVER_TEST UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(observer-warp-tests PRIVATE native_observer_minhook d3d11 dxgi bcrypt d3dcompiler)
add_library(native-input-snapshots STATIC ../bridge/input_snapshots.cpp ../bridge/output_image.cpp)
target_compile_definitions(native-input-snapshots PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(native-input-snapshots PUBLIC d3d11 d3d12 dxgi)
add_executable(input-snapshots-tests ../tests/windows/test_input_snapshots.cpp)
target_link_libraries(input-snapshots-tests PRIVATE native-input-snapshots)
if(MSVC)
  target_compile_options(native-input-snapshots PRIVATE /W4 /WX /EHsc)
  target_compile_options(input-snapshots-tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(native-input-snapshots PRIVATE -Wall -Wextra -Werror)
  target_compile_options(input-snapshots-tests PRIVATE -Wall -Wextra -Werror)
endif()
if(MSVC)
  target_compile_options(observation-tests PRIVATE /UNDEBUG)
  target_compile_options(slot-policy-tests PRIVATE /UNDEBUG)
  target_compile_options(source-pair-tests PRIVATE /UNDEBUG)
else()
  target_compile_options(observation-tests PRIVATE -UNDEBUG)
  target_compile_options(slot-policy-tests PRIVATE -UNDEBUG)
  target_compile_options(source-pair-tests PRIVATE -UNDEBUG)
endif()
add_executable(disassemble-shaders ../tools/disassemble_shaders.cpp)
target_link_libraries(disassemble-shaders PRIVATE d3dcompiler)
add_executable(flow-shader-tests ../tests/windows/test_flow_shaders.cpp)
target_compile_definitions(flow-shader-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(flow-shader-tests PRIVATE d3d11 d3dcompiler)
if(MSVC)
  target_compile_options(flow-shader-tests PRIVATE /W4 /WX /EHsc)
endif()
if(MINGW)
  target_link_options(disassemble-shaders PRIVATE -municode)
endif()
enable_testing()
add_test(NAME observation-core COMMAND observation-tests)
add_test(NAME slot-policy COMMAND slot-policy-tests)
add_test(NAME source-pair COMMAND source-pair-tests)
add_test(NAME proxy-abi COMMAND proxy-abi-tests "$<TARGET_FILE_DIR:fake-native>")
add_test(NAME observer-warp COMMAND observer-warp-tests)
add_test(NAME input-snapshots COMMAND input-snapshots-tests)
add_test(NAME flow-shaders COMMAND flow-shader-tests)
set_tests_properties(input-snapshots PROPERTIES TIMEOUT 90)
if(LS_ENABLE_NGX)
  add_test(NAME native-backend COMMAND native-backend-tests)
  add_test(NAME native-hook COMMAND native-hook-tests)
  set_tests_properties(native-backend native-hook PROPERTIES TIMEOUT 90)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/abi-fixture/$<CONFIG>/NativeDLSS.ini" CONTENT "[Observation]\nEnabled=1\n")
# The fixture's hash MUST fail the observer gate. This exercises native
# pass-through even when observation is requested for an unsupported DLL.
