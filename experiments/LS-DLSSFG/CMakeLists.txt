cmake_minimum_required(VERSION 3.24)
project(LS_DLSSFG LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(presentation_tests tests/presentation_tests.cpp)
target_include_directories(presentation_tests PRIVATE src)
add_test(NAME presentation_policy COMMAND presentation_tests)
add_executable(settings_tests tests/settings_tests.cpp)
target_include_directories(settings_tests PRIVATE src)
add_test(NAME settings_and_multiframe_policy COMMAND settings_tests)
if(NOT WIN32)
  return()
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with x64 MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(EAM_ROOT "" CACHE PATH "ls-addon-manager source")
set(NGX_ROOT "" CACHE PATH "NVIDIA/DLSS SDK source")
set(NVAPI_ROOT "" CACHE PATH "Pinned official NVIDIA NVAPI headers")
if(NOT EXISTS "${EAM_ROOT}/manager/sdk/include/eam/ihost.h" OR NOT EXISTS "${NGX_ROOT}/include/nvsdk_ngx_helpers_dlssg.h")
  message(FATAL_ERROR "Set EAM_ROOT and NGX_ROOT to the pinned SDK checkouts")
endif()
if(NOT EXISTS "${NVAPI_ROOT}/nvapi.h" OR NOT EXISTS "${NVAPI_ROOT}/nvapi_interface.h")
  message(FATAL_ERROR "Fetch pinned NVAPI headers with scripts/Fetch-NvapiHeaders.ps1 and set NVAPI_ROOT")
endif()
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 367b2c24f399988ddafc0bb4628da0106bcc09be)
FetchContent_MakeAvailable(imgui)
add_library(addon_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
target_include_directories(addon_imgui PUBLIC ${imgui_SOURCE_DIR})
add_library(LS_DLSSFG SHARED src/addon.cpp src/backend.cpp src/present_hook.cpp src/output_state.cpp src/duplicates.cpp src/optical_flow.cpp)
target_include_directories(LS_DLSSFG PRIVATE src third_party/nvof "${EAM_ROOT}/manager/sdk/include" "${NGX_ROOT}/include" "${NVAPI_ROOT}")
target_compile_definitions(LS_DLSSFG PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(LS_DLSSFG PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(LS_DLSSFG PRIVATE addon_imgui d3dcompiler d3d11 d3d12 dxgi user32 advapi32 shlwapi wintrust ole32 version bcrypt shell32
  "${NGX_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
add_executable(addon_smoke tests/addon_smoke.cpp)
target_include_directories(addon_smoke PRIVATE "${EAM_ROOT}/manager/sdk/include")
target_compile_definitions(addon_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(addon_smoke PRIVATE d3d11 dxgi user32)
add_test(NAME addon_abi COMMAND addon_smoke "$<TARGET_FILE:LS_DLSSFG>")
set_tests_properties(addon_abi PROPERTIES TIMEOUT 30)
add_executable(present_hook_tests tests/present_hook_tests.cpp src/present_hook.cpp)
target_include_directories(present_hook_tests PRIVATE src)
target_compile_definitions(present_hook_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(present_hook_tests PRIVATE d3d11 dxgi user32)
add_test(NAME present_hook_forwarding COMMAND present_hook_tests)
set_tests_properties(present_hook_forwarding PROPERTIES TIMEOUT 30)
add_executable(output_state_tests tests/output_state_tests.cpp src/output_state.cpp)
target_include_directories(output_state_tests PRIVATE src "${NVAPI_ROOT}")
target_compile_definitions(output_state_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(output_state_tests PRIVATE d3d11 dxgi user32)
add_test(NAME output_queue_restore COMMAND output_state_tests)
set_tests_properties(output_queue_restore PROPERTIES TIMEOUT 30)
add_executable(duplicate_tests tests/duplicate_tests.cpp src/duplicates.cpp)
target_include_directories(duplicate_tests PRIVATE src)
target_compile_definitions(duplicate_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(duplicate_tests PRIVATE d3d11 d3dcompiler dxgi)
add_test(NAME exact_duplicate_gpu COMMAND duplicate_tests)
set_tests_properties(exact_duplicate_gpu PROPERTIES TIMEOUT 60)
add_executable(optical_flow_tests tests/optical_flow_tests.cpp src/optical_flow.cpp)
target_include_directories(optical_flow_tests PRIVATE src third_party/nvof)
target_compile_definitions(optical_flow_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX FG_TEST_NVOF)
target_link_libraries(optical_flow_tests PRIVATE d3d11 d3dcompiler dxgi)
add_test(NAME optical_flow_gpu_contract COMMAND optical_flow_tests)
set_tests_properties(optical_flow_gpu_contract PROPERTIES TIMEOUT 60)
add_executable(interop_tests tests/interop_tests.cpp)
target_include_directories(interop_tests PRIVATE src)
target_compile_definitions(interop_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(interop_tests PRIVATE d3d11 d3d12 d3dcompiler dxgi)
add_test(NAME private_context_shared_gpu_chain COMMAND interop_tests)
set_tests_properties(private_context_shared_gpu_chain PROPERTIES TIMEOUT 30)
target_link_libraries(addon_smoke PRIVATE addon_imgui)
install(TARGETS LS_DLSSFG RUNTIME DESTINATION addons/LS_DLSSFG)
install(DIRECTORY package/ DESTINATION addons/LS_DLSSFG)
install(FILES README-RU.md NOTICE.md LICENSE LICENSE-EAM RESEARCH.md DESTINATION addons/LS_DLSSFG)
install(DIRECTORY src tests third_party DESTINATION addons/LS_DLSSFG/source)
install(FILES CMakeLists.txt DESTINATION addons/LS_DLSSFG/source)
set(NVAPI_FETCH_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/scripts/Fetch-NvapiHeaders.ps1")
set(NVAPI_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/NVIDIA-NVAPI.txt")
if(NOT EXISTS "${NVAPI_FETCH_SCRIPT}")
  set(NVAPI_FETCH_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/Fetch-NvapiHeaders.ps1")
  set(NVAPI_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/licenses/NVIDIA-NVAPI.txt")
endif()
install(FILES "${NVAPI_FETCH_SCRIPT}" DESTINATION addons/LS_DLSSFG/source/scripts)
install(FILES "${NVAPI_LICENSE}" DESTINATION addons/LS_DLSSFG)
install(FILES "${NVAPI_LICENSE}" DESTINATION addons/LS_DLSSFG/source)
