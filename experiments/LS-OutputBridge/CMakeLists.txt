cmake_minimum_required(VERSION 3.24)
project(LS_OutputBridge LANGUAGES CXX)
add_library(LS_OutputBridge SHARED src/bridge.cpp)
target_include_directories(LS_OutputBridge PUBLIC include)
target_compile_features(LS_OutputBridge PUBLIC cxx_std_17)
target_compile_definitions(LS_OutputBridge PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LS_OutputBridge PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(LS_OutputBridge PRIVATE dxgi user32)
install(TARGETS LS_OutputBridge RUNTIME DESTINATION .)
add_library(LS_SharedGpu STATIC src/output_window.cpp)
target_include_directories(LS_SharedGpu PUBLIC include)
target_compile_features(LS_SharedGpu PUBLIC cxx_std_17)
target_compile_definitions(LS_SharedGpu PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LS_SharedGpu PUBLIC user32)
