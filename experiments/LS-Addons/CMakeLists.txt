cmake_minimum_required(VERSION 3.24)
project(LS_Addons VERSION 0.3.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The addon suite requires x64 MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
add_subdirectory(../LS-OutputBridge "${CMAKE_BINARY_DIR}/output-bridge")
add_subdirectory(../LS-DLSSFG "${CMAKE_BINARY_DIR}/dlssfg")
add_subdirectory(../LS-RTXHDR "${CMAKE_BINARY_DIR}/rtxhdr")
add_subdirectory(../LS-SmoothMotion "${CMAKE_BINARY_DIR}/smooth")
add_executable(bridge_tests tests/bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE LS_OutputBridge d3d11 dxgi user32)
add_test(NAME shared_frame_order_and_lifetime COMMAND bridge_tests)
add_executable(hdr_tests tests/hdr_tests.cpp ../LS-RTXHDR/src/backend.cpp ../LS-DLSSFG/src/duplicates.cpp)
target_include_directories(hdr_tests PRIVATE ../LS-RTXHDR/src ../LS-RTXHDR/vendor/magpie ../LS-DLSSFG/src)
target_compile_definitions(hdr_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(hdr_tests PRIVATE LS_SharedGpu d3d11 d3dcompiler dxgi user32)
add_test(NAME truehdr_transport_and_tone_map COMMAND hdr_tests)
add_executable(smooth_tests tests/smooth_tests.cpp ../LS-SmoothMotion/src/backend.cpp ../LS-SmoothMotion/src/driver.cpp
  ../LS-SmoothMotion/vendor/nvsmooth30/src/nvpresent.cpp ../LS-SmoothMotion/vendor/nvsmooth30/src/fatbin.cpp ../LS-SmoothMotion/vendor/nvsmooth30/src/pe.cpp ../LS-SmoothMotion/vendor/nvsmooth30/src/profile.cpp ../LS-DLSSFG/src/duplicates.cpp)
target_include_directories(smooth_tests PRIVATE ../LS-SmoothMotion/src ../LS-SmoothMotion/vendor/nvsmooth30/include ../LS-DLSSFG/src)
target_compile_features(smooth_tests PRIVATE cxx_std_20)
target_compile_definitions(smooth_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX LS_SM_TEST)
target_link_libraries(smooth_tests PRIVATE LS_SharedGpu d3d11 d3d12 d3dcompiler dxgi dxguid user32 bcrypt)
add_test(NAME smooth_profile_and_gpu_transport COMMAND smooth_tests)
add_executable(driver_profile_tests tests/driver_profile_tests.cpp ../LS-SmoothMotion/vendor/nvsmooth30/src/profile.cpp ../LS-SmoothMotion/vendor/nvsmooth30/src/pe.cpp)
target_include_directories(driver_profile_tests PRIVATE ../LS-SmoothMotion/vendor/nvsmooth30/include)
target_compile_features(driver_profile_tests PRIVATE cxx_std_20)
target_compile_definitions(driver_profile_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME inspected_nvpresent_layout_and_controller COMMAND driver_profile_tests)
add_executable(addons_smoke tests/addons_smoke.cpp)
target_include_directories(addons_smoke PRIVATE "${EAM_ROOT}/manager/sdk/include")
target_link_libraries(addons_smoke PRIVATE addon_imgui)
add_test(NAME hdr_smooth_settings_and_lifecycle COMMAND addons_smoke "$<TARGET_FILE:LS_RTXHDR>" "$<TARGET_FILE:LS_SmoothMotion>")
set_tests_properties(shared_frame_order_and_lifetime truehdr_transport_and_tone_map smooth_profile_and_gpu_transport inspected_nvpresent_layout_and_controller hdr_smooth_settings_and_lifecycle PROPERTIES TIMEOUT 60)
install(DIRECTORY ../LS-OutputBridge ../LS-DLSSFG ../LS-RTXHDR ../LS-SmoothMotion ../LS-Addons DESTINATION source
  PATTERN ".git" EXCLUDE PATTERN "build*" EXCLUDE)
install(FILES README-RU.md Install-Addons.ps1 DESTINATION .)

install(FILES ../LS-OutputBridge/LICENSE-MIT.txt ../LS-OutputBridge/LICENSE-EAM.txt DESTINATION .)
