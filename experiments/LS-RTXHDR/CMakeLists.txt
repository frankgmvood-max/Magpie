cmake_minimum_required(VERSION 3.24)
project(LS_RTXHDR LANGUAGES CXX)
set(RTX_VIDEO_ROOT "" CACHE PATH "Pinned official RTX Video SDK 1.1.0")
if(NOT EXISTS "${RTX_VIDEO_ROOT}/include/nvsdk_ngx_helpers_truehdr.h")
  message(FATAL_ERROR "Run scripts/Fetch-RtxVideoSdk.ps1 and set RTX_VIDEO_ROOT")
endif()
add_library(LS_RtxVideoRuntime SHARED vendor/magpie/RtxVideoBridge.cpp)
target_include_directories(LS_RtxVideoRuntime PRIVATE vendor/magpie "${RTX_VIDEO_ROOT}/include")
target_compile_features(LS_RtxVideoRuntime PRIVATE cxx_std_17)
target_compile_definitions(LS_RtxVideoRuntime PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LS_RtxVideoRuntime PRIVATE "${RTX_VIDEO_ROOT}/lib/Windows/x64/nvsdk_ngx_s.lib" d3d11 dxgi shlwapi version advapi32 ole32 shell32)
add_library(LS_RTXHDR SHARED src/addon.cpp src/backend.cpp ../LS-DLSSFG/src/duplicates.cpp)
target_include_directories(LS_RTXHDR PRIVATE src vendor/magpie ../LS-DLSSFG/src "${EAM_ROOT}/manager/sdk/include")
target_compile_features(LS_RTXHDR PRIVATE cxx_std_17)
target_compile_definitions(LS_RTXHDR PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(LS_RTXHDR PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(LS_RTXHDR PRIVATE LS_OutputBridge LS_SharedGpu addon_imgui d3d11 d3dcompiler dxgi user32)
install(TARGETS LS_RTXHDR LS_RtxVideoRuntime RUNTIME DESTINATION addons/LS_RTXHDR)
install(FILES "${RTX_VIDEO_ROOT}/bin/Windows/x64/rel/nvngx_truehdr.dll" DESTINATION addons/LS_RTXHDR/runtime)
install(FILES "${RTX_VIDEO_ROOT}/NVIDIA_RTX_Video_SDK_License.pdf" DESTINATION addons/LS_RTXHDR)
install(DIRECTORY package/ DESTINATION addons/LS_RTXHDR)
install(FILES README-RU.md DESTINATION addons/LS_RTXHDR)
